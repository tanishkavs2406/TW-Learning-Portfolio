\begin{frame}{What is Beamer?}
    Beamer is a LaTeX document class for creating presentations.
    \begin{itemize}
        \item Slides are written using LaTeX.
        \item Each slide is created with a frame.
        \item The presentation can be compiled into PDF.
    \end{itemize}
\end{frame}

\begin{frame}{A Highlighted Block}
    \begin{block}{Remember}
        A frame represents one slide in a Beamer presentation.
    \end{block}
\end{frame}
